inline std::string HXD_HIDE(const std::vector<unsigned char>& bytes) {
	std::string decrypted = "";
	for (unsigned char b : bytes) {

inline void FillRGB2(int x, int y, int w, int h, D3DCOLOR Color, LPDIRECT3DDEVICE9 pDevice)
{
	D3DRECT rec = { x, y, x + w, y + h };
	pDevice->Clear(1, &rec, D3DCLEAR_TARGET, Color, 0, 0);
}

inline void DrawBorder2(int x, int y, int w, int h, int px, D3DCOLOR BorderColor, LPDIRECT3DDEVICE9 pDevice)
{
	FillRGB2(x, (y + h - px), w, px, BorderColor, pDevice);
	FillRGB2(x, y, px, h, BorderColor, pDevice);
	FillRGB2(x, y, w, px, BorderColor, pDevice);
	FillRGB2(x + w - px, y, px, h, BorderColor, pDevice);
}

inline int GetTextPixelLength(const char* text)
{
	int len = 0;
	while (*text++) len += 6;
	return len;
}

inline void DrawPixelText(int x, int y, D3DCOLOR color, const char* text, bool bBold, LPDIRECT3DDEVICE9 pDevice)
{
	if (!FontInitialized) InitSolidFont();
	int currentX = x;
	while (*text) {
		unsigned char uc = (unsigned char)(*text);
		for (int row = 0; row < 7; row++) {
			unsigned char rowData = PixelFontMask[uc][row];
			for (int col = 0; col < 5; col++) {
				if ((rowData >> (4 - col)) & 1) {
					FillRGB2(currentX + col, y + row, 1, 1, color, pDevice);
					if (bBold) FillRGB2(currentX + col + 1, y + row, 1, 1, color, pDevice);
				}
			}
		}
		currentX += bBold ? 7 : 6;
		text++;
	}
}

struct SpookyParticle {
	float x, y;
	float velocityY;
	float velocityX;
	int life;
	int maxLife;
	char displayChar;
	D3DCOLOR color;
};

inline void RenderSpookyParticles(int startX, int startY, int width, int totalHeight, LPDIRECT3DDEVICE9 pDevice) {
	static std::vector<SpookyParticle> particles;
	static int frameTicker = 0;

	frameTicker++;

	if (frameTicker % 3 == 0) {
		SpookyParticle p;
		p.x = (float)(startX + (rand() % width));
		p.y = (float)(startY + totalHeight - 2);

		p.velocityY = -0.3f - ((rand() % 10) / 30.0f);
		p.velocityX = ((rand() % 10) - 5) / 25.0f;

		p.maxLife = 90 + (rand() % 40);
		p.life = p.maxLife;

		p.displayChar = (rand() % 3 == 0) ? 'v' : '.'; 

		int colType = rand() % 4;
		if (colType == 0) p.color = D3DCOLOR_ARGB(255, 255, 120, 0); 
		else if (colType == 1) p.color = D3DCOLOR_ARGB(255, 160, 32, 240); 
		else if (colType == 2) p.color = D3DCOLOR_ARGB(255, 0, 255, 120); 
		else p.color = D3DCOLOR_ARGB(255, 200, 200, 200); 

		particles.push_back(p);
	}

	for (auto it = particles.begin(); it != particles.end();) {
		it->y += it->velocityY;
		it->x += it->velocityX;
		it->life--;

		if (it->life <= 0 || it->y < startY) {
			it = particles.erase(it);
		}
		else {
			float lifeRatio = (float)it->life / (float)it->maxLife;
			int alpha = (int)(lifeRatio * 255.0f);
			if (alpha < 0) alpha = 0;

			BYTE r = (BYTE)((it->color >> 16) & 0xFF);
			BYTE g = (BYTE)((it->color >> 8) & 0xFF);
			BYTE b = (BYTE)(it->color & 0xFF);
			D3DCOLOR fadeColor = D3DCOLOR_ARGB(alpha, r, g, b);

			char str[2] = { it->displayChar, '\0' };
			DrawPixelText((int)it->x, (int)it->y, fadeColor, str, false, pDevice);

			++it;
		}
	}

	if (particles.size() > 45) {
		particles.erase(particles.begin(), particles.begin() + 10);
	}
}

					if (ToAscii(vk, MapVirtualKey(vk, MAPVK_VK_TO_VSC), keyboardState, &translatedChar, 0) == 1) {
						char c = (char)translatedChar;
						if (c >= 32 && c <= 126) {
							EnterCriticalSection(&cs);
							if (userInputBuffer.length() < maxBufferLength) {
								userInputBuffer += c;
							}
							LeaveCriticalSection(&cs);
							Sleep(40);
						}
					}
				}
			}

			if (GetAsyncKeyState(VK_RETURN) & 1) {
				DWORD curTime = GetTickCount();
				EnterCriticalSection(&cs);
				if (!userInputBuffer.empty() && (curTime - lastClickTime > 2000)) {
					std::string finalIssue = userInputBuffer;
					userInputBuffer.clear();
					lastClickTime = curTime;

					std::thread([finalIssue]() {
						SendDiscordFeedbackSafe(finalIssue.c_str());
						}).detach();

					reportSuccessStatus = true;
				}
				LeaveCriticalSection(&cs);
				Sleep(100);
			}

			Sleep(10);
		}
	}

inline void Item(const char* text, char** opt, int& val, int maxvalue, bool bDraw, LPDIRECT3DDEVICE9 pDevice)
{
	if (!bDraw) return;

	if (val < 0) val = 0;
	if (val >= maxvalue) val = maxvalue - 1;

	int itemIndex = qntHacks;
	if (MenuSelection == itemIndex)
	{
		if ((GetAsyncKeyState(VK_RIGHT) & 1) && (val < maxvalue - 1)) val++;
		if ((GetAsyncKeyState(VK_LEFT) & 1) && (val > 0)) val--;
	}

	if (pDevice != NULL) {
		int textY = menuy + 34 + (itemIndex * itemHeight);
		bool isSelected = (MenuSelection == itemIndex);

		D3DCOLOR itemTextColor = isSelected ? UI_COLOR_TEXT_ACT : UI_COLOR_TEXT_PAS;
		D3DCOLOR valueColor = (val > 0) ? UI_STATE_ON : UI_STATE_OFF;
		D3DCOLOR squareColor = isSelected ? UI_COLOR_TEXT_ACT : ((val > 0) ? UI_STATE_ON : UI_COLOR_TEXT_WHITE);

		FillRGB2(menux + padding + 2, textY + 7, 3, 3, squareColor, pDevice);
		DrawPixelText(menux + padding + 10, textY + 4, itemTextColor, text, false, pDevice);

		char valueBuffer[64];
		sprintf_s(valueBuffer, ": [%s]", opt[val]);

		int valueXAnchor = menux + menuW - (GetTextPixelLength(valueBuffer) + padding);
		DrawPixelText(valueXAnchor, textY + 4, valueColor, valueBuffer, isSelected, pDevice);
	}

	qntHacks++;
}

inline void Folder(const char* text, int& val, bool bDraw, bool isLocked, LPDIRECT3DDEVICE9 pDevice)

	if (val < 0) val = 0;
	if (val > 1) val = 1;

	int itemIndex = qntHacks;

	{
		if ((GetAsyncKeyState(VK_RIGHT) & 1) || (GetAsyncKeyState(VK_LEFT) & 1)) val = !val;
	}

	if (isLocked) val = 0;

	if (pDevice != NULL) {
		int textY = menuy + 34 + (itemIndex * itemHeight);

		D3DCOLOR barBgColor = isSelected ? UI_FOLDER_BG_ACT : UI_FOLDER_BG_PAS;

		DrawPixelText(indicatorXAnchor, textY + 4, indicatorColor, indicator, false, pDevice);
	}

	qntHacks++;
}

	Item("Anti-Kick G", Opt_OnOff, MenuItem[56], 2, (MenuItem[503] == 1 && bDraw), pDevice);
	Item("Anti-Kick R", Opt_OnOff, MenuItem[57], 2, (MenuItem[503] == 1 && bDraw), pDevice);
	Item("Game Mode", Opt_type, MenuItem[78], 10, (MenuItem[503] == 1 && bDraw), pDevice);
	Item("Loadout", Opt_LoadoutAction, loadoutActionVal, 3, (MenuItem[503] == 1 && bDraw), pDevice);
	if (loadoutActionVal == 1) {
		SaveLoadoutConfig("phantomx_loadout.ini");
		loadoutActionVal = 0;
	}
	else if (loadoutActionVal == 2) {
		LoadLoadoutConfig("phantomx_loadout.ini");
		loadoutActionVal = 0;
	}

	Folder("MENU", MenuItem[803], bDraw, false, pDevice);
	Item("Key", (char**)Opt_key_Items, Opt_key, 21, (MenuItem[803] == 1 && bDraw), pDevice);
	CurrentMaxItems = qntHacks;
}

inline void Navigation(LPDIRECT3DDEVICE9 pDevice)
{
	if (Menu)
	{
		Itens(true, NULL);

		if (GetAsyncKeyState(VK_DOWN) & 1) MenuSelection++;
		if (GetAsyncKeyState(VK_UP) & 1) MenuSelection--;

		if (CurrentMaxItems > 0) {
			if (MenuSelection >= CurrentMaxItems) MenuSelection = 0;
			if (MenuSelection < 0) MenuSelection = CurrentMaxItems - 1;
		}
		else {
			MenuSelection = 0;
		}
	}
}

inline void RenderMenu(LPDIRECT3DDEVICE9 pDevice)
{
	bool bKeyPressed = false;

	switch (Opt_key)
	{
	case 0: if (GetAsyncKeyState(VK_INSERT) & 1) bKeyPressed = true; break;
	case 1: if ((GetAsyncKeyState(VK_CONTROL) & 0x8000) && (GetAsyncKeyState(VK_INSERT) & 1)) bKeyPressed = true; break;
	case 2: if ((GetAsyncKeyState(VK_MENU) & 0x8000) && (GetAsyncKeyState(VK_INSERT) & 1)) bKeyPressed = true; break;
	case 3: if ((GetAsyncKeyState(VK_SHIFT) & 0x8000) && (GetAsyncKeyState(VK_INSERT) & 1)) bKeyPressed = true; break;
	case 4: if ((GetAsyncKeyState(VK_CONTROL) & 0x8000) && (GetAsyncKeyState(VK_DELETE) & 1)) bKeyPressed = true; break;
	case 5: if ((GetAsyncKeyState(VK_MENU) & 0x8000) && (GetAsyncKeyState(VK_DELETE) & 1)) bKeyPressed = true; break;
	case 6: if (GetAsyncKeyState(VK_HOME) & 1) bKeyPressed = true; break;
	case 7: if (GetAsyncKeyState(VK_END) & 1) bKeyPressed = true; break;
	case 8: if (GetAsyncKeyState(VK_PRIOR) & 1) bKeyPressed = true; break;
	case 9: if (GetAsyncKeyState(VK_NEXT) & 1) bKeyPressed = true; break;
	case 10: if (GetAsyncKeyState(VK_F1) & 1) bKeyPressed = true; break;
	case 11: if (GetAsyncKeyState(VK_F2) & 1) bKeyPressed = true; break;
	case 12: if (GetAsyncKeyState(VK_F3) & 1) bKeyPressed = true; break;
	case 13: if (GetAsyncKeyState(VK_F4) & 1) bKeyPressed = true; break;
	case 14: if (GetAsyncKeyState(VK_F5) & 1) bKeyPressed = true; break;
	case 15: if (GetAsyncKeyState(VK_F6) & 1) bKeyPressed = true; break;
	case 16: if (GetAsyncKeyState(VK_F7) & 1) bKeyPressed = true; break;
	case 17: if (GetAsyncKeyState(VK_F8) & 1) bKeyPressed = true; break;
	case 18: if ((GetAsyncKeyState(VK_CONTROL) & 0x8000) && (GetAsyncKeyState(VK_SHIFT) & 0x8000) && (GetAsyncKeyState(VK_INSERT) & 1)) bKeyPressed = true; break;
	case 19: if ((GetAsyncKeyState(VK_CONTROL) & 0x8000) && (GetAsyncKeyState(VK_MENU) & 0x8000) && (GetAsyncKeyState(VK_INSERT) & 1)) bKeyPressed = true; break;
	case 20: if ((GetAsyncKeyState(VK_CONTROL) & 0x8000) && (GetAsyncKeyState(VK_SHIFT) & 0x8000) && (GetAsyncKeyState(VK_HOME) & 1)) bKeyPressed = true; break;
	}

	if (bKeyPressed) Menu = !Menu;

	if (Menu)
	{
		static DWORD lastLockCheck = 0;
		if (lastLockCheck == 0 || (GetTickCount() - lastLockCheck > 15000)) {
			lastLockCheck = GetTickCount();
			std::thread([]() { MenuLocks::FetchLocks(); }).detach();
		}

		Navigation(pDevice);

		int topCapHeight = 35;
		int footerHeight = 28;
		int totalMenuHeight = (CurrentMaxItems * itemHeight) + topCapHeight + footerHeight;

		g_Changelog.UpdateAndRender(menux, menuy, menuW, pDevice);
		g_SupportPanel.UpdateAndRender(menux, menuy, menuW, pDevice);

		pDevice->SetRenderState(D3DRS_ALPHABLENDENABLE, TRUE);
		pDevice->SetRenderState(D3DRS_SRCBLEND, D3DBLEND_SRCALPHA);
		pDevice->SetRenderState(D3DRS_DESTBLEND, D3DBLEND_INVSRCALPHA);

		FillRGB2(menux, menuy, menuW, totalMenuHeight, UI_COLOR_BG, pDevice);

		static int lightAnim = 0;
		lightAnim++;
		D3DCOLOR SpookyLight1 = (lightAnim % 40 < 20) ? D3DCOLOR_ARGB(255, 255, 120, 0) : D3DCOLOR_ARGB(255, 160, 32, 240);
		D3DCOLOR SpookyLight2 = (lightAnim % 40 < 20) ? D3DCOLOR_ARGB(255, 160, 32, 240) : D3DCOLOR_ARGB(255, 0, 255, 120);

		int bLen = 8;
		FillRGB2(menux + 4, menuy + 4, bLen, 1, SpookyLight1, pDevice);
		FillRGB2(menux + 4, menuy + 4, 1, bLen, SpookyLight1, pDevice);

		FillRGB2(menux + menuW - 4 - bLen, menuy + 4, bLen, 1, SpookyLight2, pDevice);
		FillRGB2(menux + menuW - 5, menuy + 4, 1, bLen, SpookyLight2, pDevice);

		FillRGB2(menux + 4, menuy + totalMenuHeight - 5, bLen, 1, SpookyLight2, pDevice);
		FillRGB2(menux + 4, menuy + totalMenuHeight - 4 - bLen, 1, bLen, SpookyLight2, pDevice);

		FillRGB2(menux + menuW - 4 - bLen, menuy + totalMenuHeight - 5, bLen, 1, SpookyLight1, pDevice);
		FillRGB2(menux + menuW - 5, menuy + totalMenuHeight - 4 - bLen, 1, bLen, SpookyLight1, pDevice);

		DrawBorder2(menux, menuy, menuW, totalMenuHeight, 2, UI_COLOR_FRAME, pDevice);
		DrawBorder2(menux + 2, menuy + 2, menuW - 4, totalMenuHeight - 4, 1, UI_COLOR_INNER, pDevice);

		DrawSpookyHeader(menux, menuy, menuW, pDevice);

		InitializeMenuStrings();

		const char* brandText = safeBrandText.c_str();
		int brandX = menux + ((menuW - GetTextPixelLength(brandText)) / 2);
		DrawPixelText(brandX, menuy + 8, D3DCOLOR_ARGB(255, 255, 120, 0), brandText, true, pDevice); 

		const char* subBrandText = "PHANTOM x RAGNARX";
		int subBrandX = menux + ((menuW - GetTextPixelLength(subBrandText)) / 2);
		DrawPixelText(subBrandX, menuy + 17, D3DCOLOR_ARGB(255, 0, 255, 120), subBrandText, false, pDevice); 

		FillRGB2(menux + 8, menuy + 26, menuW - 16, 1, D3DCOLOR_ARGB(255, 255, 120, 0), pDevice);
		FillRGB2(menux + 12, menuy + 27, menuW - 24, 1, D3DCOLOR_ARGB(200, 80, 20, 110), pDevice);

		Itens(true, pDevice);

		int footerY = menuy + totalMenuHeight - footerHeight + 2;
		FillRGB2(menux + 10, footerY - 2, menuW - 20, 1, UI_COLOR_INNER, pDevice);

		static float framesPerSecond = 0.0f;
		static int frameCount = 0;
		static DWORD lastTime = 0;
		DWORD currentTime = GetTickCount();
		frameCount++;

		if (currentTime - lastTime >= 1000) {
			framesPerSecond = (float)frameCount * 1000.0f / (float)(currentTime - lastTime);
			frameCount = 0;
			lastTime = currentTime;
		}

		std::time_t t = std::time(nullptr);
		std::tm now; localtime_s(&now, &t);

		char dateBuffer[32];  std::strftime(dateBuffer, sizeof(dateBuffer), "%m/%d", &now);
		char timeBuffer[32];  std::strftime(timeBuffer, sizeof(timeBuffer), "%H:%M", &now);

		char dateTimeStr[64];
		sprintf_s(dateTimeStr, "%s | %s", dateBuffer, timeBuffer);
		int dtX = menux + ((menuW - GetTextPixelLength(dateTimeStr)) / 2);
		DrawPixelText(dtX, footerY + 2, UI_COLOR_TEXT_WHITE, dateTimeStr, false, pDevice);

		char fpsStr[32];
		sprintf_s(fpsStr, "LIVE FPS: %.1f", framesPerSecond);
		int fpsX = menux + ((menuW - GetTextPixelLength(fpsStr)) / 2);
		DrawPixelText(fpsX, footerY + 12, UI_STATE_ON, fpsStr, true, pDevice);

		RenderSpookyParticles(menux, menuy, menuW, totalMenuHeight, pDevice);
	}
}
